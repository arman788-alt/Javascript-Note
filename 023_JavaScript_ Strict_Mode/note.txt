জাভাস্ক্রিপ্টে Strict Mode হলো এমন একটি বিশেষ মোড যা তোমার কোডকে আরও নিরাপদ এবং ভুলমুক্ত করতে সাহায্য করে। এটি জাভাস্ক্রিপ্ট ইঞ্জিনকে বলে দেয়— "কোডে কোনো আলসেমি বা ভুল সহ্য করা হবে না, সবকিছু নিয়ম মেনে করতে হবে।"

সহজ কথায়, সাধারণ জাভাস্ক্রিপ্টে অনেক ভুল করলেও ব্রাউজার কিছু বলে না, কিন্তু Strict Mode অন করলে সেই ভুলগুলোর জন্য সে সরাসরি এরর (Error) দিয়ে দিবে।

১. কীভাবে অন করতে হয়?
কোডের একদম শুরুতে শুধু একটি স্ট্রিং লিখে দিলেই এটি চালু হয়ে যায়:

"use strict";

// এখন থেকে পুরো কোডটি Strict Mode এ চলবে
তুমি চাইলে এটি কোনো নির্দিষ্ট ফাংশনের ভেতরেও ব্যবহার করতে পারো। তখন শুধু ওই ফাংশনটি কড়া নিয়মে চলবে।


২. কেন আমরা Strict Mode ব্যবহার করি? (৫টি প্রধান কারণ)

ক) অঘোষিত ভ্যারিয়েবল আটকানো (No more Auto-Global)
আগে আমরা দেখেছি let/var ছাড়া ভ্যারিয়েবল লিখলে তা গ্লোবাল হয়ে যায়। Strict Mode এটি হতে দেয় না।

"use strict";
name = "Arman"; // Error! name is not defined


খ) প্যারামিটারের নাম ডুপ্লিকেট করা নিষেধ
সাধারণ মোডে ফাংশনের দুইটা প্যারামিটারের নাম এক হলে জাভাস্ক্রিপ্ট মাইন্ড করে না, কিন্তু Strict Mode এ এটি বড় অপরাধ।

"use strict";
function sum(a, a, b) { // Error! Duplicate parameter name not allowed
  return a + a + b;
}


গ) রিড-অনলি প্রপার্টিতে লিখতে বাধা দেয়
এমন কিছু প্রপার্টি আছে যা শুধু পড়া যায় কিন্তু পরিবর্তন করা যায় না। Strict Mode এ সেগুলো পরিবর্তনের চেষ্টা করলে এরর দিবে।


"use strict";
const obj = {};
Object.defineProperty(obj, "x", {value: 10, writable: false});
obj.x = 20; // Error! (সাধারণ মোডে এটি এরর দেয় না, কিন্তু কাজও করে না)


ঘ) this এর নিরাপত্তা (Secure this)
সাধারণ ফাংশনে this কল করলে তা window অবজেক্টকে দেখায়। কিন্তু Strict Mode এ সেটি নিরাপত্তা খাতিরে undefined দেখাবে।

"use strict";
function show() {
  console.log(this); 
}
show(); // Output: undefined


ঙ) ভবিষ্যতের জন্য সংরক্ষিত শব্দ (Reserved Words)
জাভাস্ক্রিপ্টের ভবিষ্যতের ভার্সনে যে শব্দগুলো কিউওয়ার্ড হিসেবে আসতে পারে (যেমন: interface, package, private), সেগুলো তুমি ভ্যারিয়েবল হিসেবে ব্যবহার করতে পারবে না।
