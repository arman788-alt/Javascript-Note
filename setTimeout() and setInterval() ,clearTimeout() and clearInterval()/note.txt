JavaScript-এ সময়ের সাথে কোনো কাজ নিয়ন্ত্রণ করার জন্য এই ফাংশনগুলো ব্যবহার করা হয়। এগুলো মূলত Asynchronous কাজ করার জন্য বেশ জনপ্রিয়।সহজ ভাষায় নিচে এদের কাজ এবং পার্থক্য বুঝিয়ে দিচ্ছি:

১. setTimeout() এবং clearTimeout() setTimeout() ব্যবহার করা হয় যখন আপনি চান কোনো একটি কাজ নির্দিষ্ট সময় পর মাত্র একবার হবে।কিভাবে কাজ করে: আপনি সময় সেট করে দেবেন (মিলি-সেকেন্ডে), সেই সময় পার হওয়ার পর ফাংশনটি চলবে।

clearTimeout(): যদি সময় শেষ হওয়ার আগেই আপনি পরিকল্পনা পরিবর্তন করেন এবং কাজটি থামাতে চান, তবে এটি ব্যবহার করা হয়।

উদাহরণ:
JavaScript// ৩ সেকেন্ড (৩০০০ মিলি-সেকেন্ড) পর মেসেজ দেখাবে

const timerId = setTimeout(() => {
    console.log("হ্যালো! আমি ৩ সেকেন্ড পর আসলাম।");
}, 3000);

// যদি কোনো কারণে কাজটা বাতিল করতে চান:
// clearTimeout(timerId); 



২. setInterval() এবং clearInterval() setInterval() ব্যবহার করা হয় যখন আপনি চান কোনো একটি কাজ নির্দিষ্ট সময় পর পর বারবার হতে থাকবে।কিভাবে কাজ করে: এটি একবার শুরু হলে চলতেই থাকে যতক্ষণ না আপনি একে থামিয়ে দিচ্ছেন।clearInterval(): এটি ব্যবহার করে চলমান লুপ বা ইন্টারভালটিকে থামানো হয়। এটি না করলে আপনার কোড সারাজীবন চলতেই থাকবে, যা মেমোরির জন্য ক্ষতিকর।

উদাহরণ:JavaScriptlet count = 0;

const intervalId = setInterval(() => {
    count++;
    console.log("সেকেন্ড চলছে: " + count);

    // ৫ সেকেন্ড পর অটোমেটিক বন্ধ করে দেবে
    if (count === 5) {
        clearInterval(intervalId);
        console.log("ইন্টারভাল বন্ধ করা হলো।");
    }
}, 1000);
