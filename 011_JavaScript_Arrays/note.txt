// splice() দিয়ে Array-এর মাঝখান থেকে
// Delete / Add / Replace করা যায়।

// Syntax:
// array.splice(start, deleteCount, item1, item2, ...);


// Add items
language.splice(2, 0, "info", "info2");


// Delete items
language.splice(2, 2);


console.log(language);


// Example:
//
// splice(2, 0, "info", "info2")
//
// 2       → index 2 থেকে শুরু
// 0       → 0টি item delete
// info    → add
// info2   → add


// splice(2, 2)
//
// 2 → index 2 থেকে শুরু
// 2 → 2টি item delete

// 1 → শুরু index

// 18. Deep / Multi-level Nested Array
// ============================================================

const deepArray = [
    1,
    [
        2,
        [
            3,
            [4]
        ]
    ]
];


// ============================================================
// 19. flat(1)
// ============================================================

// ১ level ভাঙবে

console.log(deepArray.flat(1));

// Output:
// [1, 2, [3, [4]]]


// ============================================================
// 20. flat(2)
// ============================================================

// ২ level ভাঙবে

console.log(deepArray.flat(2));

// Output:
// [1, 2, 3, [4]]


// ============================================================
// 21. flat(3)
// ============================================================

// ৩ level ভাঙবে

console.log(deepArray.flat(3));

// Output:
// [1, 2, 3, 4]


// ============================================================
// 22. flat(Infinity)
// ============================================================

// যত গভীর Nested Array থাকুক,
// সব level ভেঙে ফেলবে।

console.log(deepArray.flat(Infinity));

// Output:
// [1, 2, 3, 4]
